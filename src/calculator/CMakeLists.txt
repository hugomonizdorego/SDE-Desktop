pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.8)
pkg_check_modules(JSC REQUIRED IMPORTED_TARGET javascriptcoregtk-4.1)
string(JSON SDE_CALCULATOR_VERSION GET "${SDE_APP_VERSIONS}" calculator)
configure_file(calculator_version.h.in ${CMAKE_CURRENT_BINARY_DIR}/calculator_version.h @ONLY)
set(CALCULATOR_ENGINE "${SDE_COMPONENT_SOURCES}/calculator/js/engine.js")
if(NOT EXISTS "${CALCULATOR_ENGINE}")
  message(FATAL_ERROR "Run python3 tools/prepare-components.py before configuring Calculator")
endif()
file(READ "${CALCULATOR_ENGINE}" CALCULATOR_ENGINE_SOURCE)
configure_file(calculator_engine.h.in ${CMAKE_CURRENT_BINARY_DIR}/calculator_engine.h @ONLY)
add_executable(calculator main.cpp)
target_include_directories(calculator PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(calculator PRIVATE PkgConfig::GTK4 PkgConfig::JSC)
install(TARGETS calculator RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES "${CALCULATOR_ENGINE}" DESTINATION ${CMAKE_INSTALL_DATADIR}/sde/calculator)
install(FILES "${SDE_COMPONENT_SOURCES}/calculator/LICENSE" "${SDE_COMPONENT_SOURCES}/calculator/copyright"
        DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/sde/upstream/scientific-calculator)
